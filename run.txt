--- B1. Cap nhat ma nguon ----------------------------------------------------

  Set-Location D:\pre-huspm-adaptive

--- B2. Dot do THOI GIAN (mem_mode = used) -----------------------------------

  # run.ps1 tu bien dich, tu ghi manifest, tu kiem tra may va heap truoc khi chay.
  # -ExpectMachine bat script tu choi neu lo chay nham may khac.
  # -Dseeds va -Drepeats truyen qua JAVA_TOOL_OPTIONS vi run.ps1 khong co co rieng.
  #   repeats = so lan lap CUNG mot cau hinh, de lay do lech chuan (nhieu JIT/GC)
  #   seeds   = so cau hinh khac nhau (cach chia lo, chuoi thao tac)
  # Luoi day du la seeds=5, repeats=3.
  $env:JAVA_TOOL_OPTIONS = '-Dseeds=5 -Drepeats=3'
  .\run.ps1 -Heap 24g -Stack 64m -TimeoutMin 120 -ExpectMachine 'Admin-PC' SC1 SC2 SC3 SC4 SC5 SC6 SC7
  Remove-Item Env:\JAVA_TOOL_OPTIONS

  # Ket qua vao  results\run-<thoi diem>-<commit>\
  # Hinh     vao  figures\run-<thoi diem>-<commit>\


--- B3. Dot do BO NHO (mem_mode = live) --------------------------------------

  # Chay RIENG, sau khi B2 xong han. Che do live ep thu gom nen thoi gian cua
  # dot nay vo nghia (runtime_ms = -3) - no chi de lay bo nho.
  # Launcher tu lap qua tung thuat toan, MOI THUAT TOAN MOT JVM RIENG.
  # repeats=1 o dot bo nho: tap song gan nhu tat dinh nen lap lai khong thu duoc
  # them thong tin, trong khi ep GC da lam dot nay dat hon dot do thoi gian nhieu.
  # Seed van giu day du vi no doi CAU HINH chu khong phai do nhieu.
  $env:JAVA_TOOL_OPTIONS = '-Dseeds=5 -Drepeats=1'
  .\run.ps1 -Heap 24g -Stack 64m -TimeoutMin 120 -MemMode live -MemIntervalMs 1000 -ExpectMachine 'Admin-PC' SC1 SC2 SC3 SC4 SC5 SC6 SC7
  Remove-Item Env:\JAVA_TOOL_OPTIONS

  # Ket qua vao  results\run-<thoi diem>-<commit>\mem\


--- B4. Kiem tra ket qua truoc khi push --------------------------------------

  $run = Get-ChildItem results -Directory -Filter 'run-*' | Sort-Object Name | Select-Object -Last 1
  $run.FullName
  Get-ChildItem $run.FullName -Recurse -File | Select-Object Name, Length

  # Manifest phai ghi dung may va dung heap:
  Get-Content "$($run.FullName)\manifest.json" | ConvertFrom-Json |
      Select-Object machine_name, ram_gb, cpu, jvm_flags, timeout_minutes, mem_mode, java_version

  # Trung binh va do lech chuan tren cac lan lap:
  python analysis\aggregate.py $run.FullName --csv "$($run.FullName)\aggregate.csv"

  # Dem so dong va so ca that bai cua tung file ket qua:
  Get-ChildItem $run.FullName -Recurse -Filter *.csv | ForEach-Object {
      $rows = Import-Csv $_.FullName
      [PSCustomObject]@{
          File = $_.Name
          Rows = $rows.Count
          OT   = ($rows | Where-Object { $_.op_type -like '*_OT*'  }).Count
          OOM  = ($rows | Where-Object { $_.op_type -like '*_OOM*' }).Count
          Fail = ($rows | Where-Object { $_.op_type -like '*_FAILED*' }).Count
      }
  } | Format-Table -AutoSize

===============================================================================
 PHU LUC - CAC CO CO THE TRUYEN
===============================================================================

  run.ps1
    -Heap 24g              kich thuoc heap; script tu choi neu lon hon RAM - 4 GB
    -Stack 64m             stack moi thread (de quy sau can cai nay)
    -TimeoutMin 120        gioi han moi loi goi, phut
    -MemMode used|live     used = do thoi gian; live = do tap song
    -MemIntervalMs 1000    khoang ep thu gom o che do live
    -MemArms @('...')      danh sach thuat toan cho che do live
    -ExpectMachine 'Ten'   tu choi neu chay nham may
    -AllowConcurrent       bo qua kiem tra "da co java dang chay" (chi dung khi
                           KHONG phai lan chay do)

  Thuoc tinh JVM (truyen bang -D... khi goi java truc tiep)
    -Dtimeout.min=120         gioi han moi loi goi
    -Dmem.mode=live           che do do bo nho
    -Dmem.interval.ms=1000    khoang ep thu gom
    -Dalgo.only=Adaptive-Full chi chay mot thuat toan
    -Ddataset.only=SIGN       chi chay mot dataset
    -Dadapt.trigger=iaumax    luat kich hoat rescan: swumax | cub | iaumax
    -Dblock.size=100          B_sz cua chi muc nguoc cap do khoi
    -Dkappa=0.5               he so thich ung
    -Ddelta0=1e-4             bien do khoi tao
    -Drepeats=3               so lan lap CUNG mot cau hinh (do nhieu JIT/GC)
    -Dseeds=5                 so cau hinh khac nhau. Chi co tac dung o SC2 (khi >1),
                              SC5 chien luoc BURST, va SC6. Con lai tat dinh.
    -Dresults.dir=...         thu muc ket qua (che do live bat buoc co 'mem')

  Ten thuat toan dung cho -MemArms va -Dalgo.only
    USpan | HUSP-ULL | HUSP-SP | IncUSP-Miner+ | Pre-HUSPM | Pre-HUSPM-TSU | Adaptive-Full

    HUSP-SP la thuat toan TINH, chi co mat o SC1 va SC3. Dua no vao -MemArms cho
    cac kich ban gia tang se chi sinh ra file rong.

 XU LY SU CO
===============================================================================

  "Heap 24g does not fit in ... GB of RAM"
      Dang chay nham may. Kiem tra (Get-CimInstance Win32_ComputerSystem).Name

  "java process(es) already running"
      Con JVM cua dot truoc. Doi no xong, hoac:
          Get-Process java | Stop-Process -Force

  "-Dmem.mode=live phai ghi vao thu muc co chua 'mem'"
      Dang goi java truc tiep o che do live. Dung run.ps1 -MemMode live, hoac
      them 'mem' vao -Dresults.dir.

  "che do live yeu cau MOI ARM MOT JVM RIENG"
      Thieu -Dalgo.only. Dung run.ps1 -MemMode live de no tu lap.

  javac bao "invalid flag"
      Argfile bi ghi kem BOM. run.ps1 da xu ly; neu tu go tay thi phai dung
      UTF8Encoding($false) nhu o muc A6.

  Chay giua chung bi mat dien
      Ket qua da ghi den dau con den do (CSV duoc flush theo tung dong). Chay
      lai kich ban con thieu, moi lan chay vao mot thu muc rieng nen khong de.
